\part*{APPENDICES}

\chapter{Proof of Concept - Stan Gry}
\label{appA}

Ten dodatek pokazuje, w jakim stanie znajduje się gra w momencie składania pracy. Jest to żywy build, nie projekt na papierze - każdy z opisanych systemów uruchamia się w realnej sesji, a większość zrzutów ekranu pochodzi z mojego rozegrania w trakcie końcowych testów regresji.

\section{Parametry Techniczne}

\begin{itemize}
    \item \textbf{Silnik}: Godot 4.x, renderer Forward Plus,
    \item \textbf{Język}: C\# (.NET 8),
    \item \textbf{Rozdzielczość}: 1920$\times$1080,
    \item \textbf{Platforma}: Windows (PC),
    \item \textbf{Pliki kodu}: ponad 70 plików \texttt{.cs} w katalogu \texttt{Scripts/},
    \item \textbf{Sceny Godot}: ponad 30 plików \texttt{.tscn} w katalogu \texttt{Scenes/},
    \item \textbf{Serializacja zapisu}: \texttt{System.Text.Json},
    \item \textbf{Kontrola wersji}: Git (repozytorium lokalne).
\end{itemize}

\section{Zrealizowane Mechaniki}

\subsection{Hub Świata (Katakumby)}
Lokacja startowa stanowiąca hub gry. Mroczne korytarze z oświetleniem punktowym, NPC-dawca questa (Labrador Rycerz), brama wejściowa do lochu. Środowisko 3D renderowane w 2.5D.

\subsection{Proceduralny Loch}
Trzy warianty układu (różna liczba pokoi i paleta świateł) generowane algorytmem \textit{drunkard's walk} na siatce $9 \times 9$. Losowane modyfikatory runu: RichCoins, HeavyElites, Gloom. Spawny wrogów z wspólnym balansem HP/siły, elit i bossa w pokoju \texttt{Boss}, do trzech skrzyń, 1-2 radarów, portal wyjścia w pokoju startowym. Klucz questowy z \texttt{LockedCabinet} w hubie; relikt przy pierwszym wyjściu z lochu.

\subsection{Postać Gracza}
\begin{itemize}
    \item Ruch w czterech kierunkach (WASD),
    \item Atak (Spacja) z hitboxem i obrażeniami,
    \item Dash/unik (Shift),
    \item Statystyki: HP, Strength, Gold, EXP,
    \item System ekwipunku: 20 slotów inventory + 5 slotów wyposażenia,
    \item Ulepszanie przedmiotów przez scalanie A+B (do \texttt{+MaxUpgradeLevel}),
    \item Ekran śmierci po utracie HP.
\end{itemize}

\subsection{Przeciwnicy}
\begin{itemize}
    \item \textbf{Zwykły wróg}: pełna maszyna stanów AI (idle, patrol, pościg, atak, ogłuszenie, śmierć),
    \item \textbf{Elita}: rozszerzona wersja z hive-mindem \texttt{EliteBrain} (kontekstowy bandyta, predykcja ruchu, shaderowy tint fioletu),
    \item \textbf{Boss}: wróg o skali 1,5$\times$ z czerwonym tintem, dashem i bombami obszarowymi (tylko w \texttt{ChaseArea}, \texttt{TargetsPlayerOnly}) w pokoju \texttt{Boss},
    \item \textbf{Niewidzialny}: wróg ukryty do momentu ujawnienia przez radar (3~s w polu wykrywania); na podłodze rysuje subtelny okrągły cień,
    \item System dropów z pulą konsumowalnych i pozostałych przedmiotów,
    \item Animacja śmierci, usunięcie z planszy, aktualizacja licznika w HUD.
\end{itemize}

\subsection{Radar i Wykrywanie Niewidzialnych}
\begin{itemize}
    \item Półkulista strefa wykrywania (\texttt{SphereMesh} z \texttt{is\_hemisphere=true}, promień 8~m),
    \item Płaska podstawa półkuli leżąca na podłodze; korzeń radaru zakopany 5~cm pod warstwą tekstury podłogi,
    \item Detekcja per-klatka fizyki przez \texttt{Area3D.GetOverlappingBodies()} z \texttt{HashSet}-em ,,kto był w środku w poprzedniej klatce",
    \item Próg ujawnienia 3~s; po przekroczeniu \texttt{Reveal()} przełącza shader na pełną widoczność i emituje toast,
    \item Spawn 1-2 radarów na loch w losowych pokojach typu \texttt{Normal}/\texttt{Treasure},
    \item Sprite radaru z obrotową misą (\texttt{DishRotationSpeed = 1{,}2}~rad/s).
\end{itemize}

\subsection{Minimapa}
\begin{itemize}
    \item Renderowanie w \texttt{SubViewport} (\texttt{MinimapSize}~$\times$~\texttt{MinimapSize}~px),
    \item Wyświetlanie przez \texttt{TextureRect} ze shaderem \texttt{canvas\_item} liczącym maskę kołową \texttt{1 - smoothstep(0.395, 0.412, d)},
    \item Sprite'y pokoi dobierane na podstawie konfiguracji drzwi (5 grafik źródłowych $\times$ 4 obroty = 16 kombinacji),
    \item Dynamiczne ikony znajdziek (skrzynie, monety, portal startowy) odświeżane co klatkę,
    \item Kropki przeciwników z rozróżnieniem typu po grupie Godot (zwykli 12~px, elity 13~px, niewidzialni 13~px, boss 16~px),
    \item Tryb \textit{Gloom}: \texttt{Modulate (0,48; 0,50; 0,56)} dla mapy i przygaszona kropka gracza.
\end{itemize}

\subsection{Adaptacyjna AI (\texttt{EliteBrain})}
\begin{itemize}
    \item Contextual multi-armed bandit $\varepsilon$-greedy (4 taktyki $\times$ 9 kontekstów = 36 ramion),
    \item On-line Q-learning z EMA ($\alpha = 0{,}12$, $\varepsilon_0 = 0{,}35$, $\varepsilon_\text{floor} = 0{,}05$),
    \item Model gracza (EMA prędkości, wycofywania, DPS),
    \item Predykcja pozycji gracza z adaptacyjnym \texttt{LeadTime},
    \item Adaptacja \texttt{Aggression}, \texttt{Caution}, \texttt{PreferredDistance},
    \item Persystencja stanu w \texttt{user://brain.json} z wersjonowaniem schematu,
    \item Debug HUD (\texttt{F3}) pokazujący aktualny stan AI.
\end{itemize}

\subsection{System Questów}
\begin{itemize}
    \item Quest poboczny \textit{Clean the Storage} (Kill $N$ wrogów \texttt{Roach}) od NPC startowego,
    \item Sekwencyjny quest główny \textit{Pętla Vanitas} (klucz $\rightarrow$ głębia $\rightarrow$ relikt $\rightarrow$ rozmowa ze strażnikiem),
    \item Typy celów: Kill, Collect, ReachArea, Talk,
    \item Menu questów z listą aktywnych/ukończonych i panelem szczegółów,
    \item Dziennik z narracyjnymi wpisami po każdym kroku questa głównego,
    \item Nagrody (złoto, EXP, przedmioty) po ukończeniu.
\end{itemize}

\subsection{Mini-Gry (5 typów)}
\begin{itemize}
    \item \textbf{Lockpick} - łamanie zamka przez łapanie ruchomego wskaźnika,
    \item \textbf{PipePuzzle} - łączenie rur rotacją segmentów,
    \item \textbf{BlockFit} - układanie kształtów w dopasowania,
    \item \textbf{BlockMaze} - przesuwanie klocka po labiryncie,
    \item \textbf{LightSequence} - zapamiętywanie i odtwarzanie sekwencji kolorów.
\end{itemize}

\subsection{System Zapisu Gry}
\begin{itemize}
    \item Autosave w formacie JSON (\texttt{user://autosave.json}),
    \item Wersjonowanie schematu (\texttt{SaveVersion}),
    \item Snapshot ulepszeń przedmiotów w \texttt{InvSlotDto},
    \item Osobny plik stanu AI (\texttt{user://brain.json}) z wersjonowaniem,
    \item Automatyczny save: po wejściu/wyjściu z lochu, po ukończeniu questa, po zapisie w strefie bezpiecznej,
    \item Ekran startowy z opcjami \textit{Nowa gra} / \textit{Kontynuuj}.
\end{itemize}

\subsection{Interfejs Użytkownika}
\begin{itemize}
    \item \textbf{HUD}: pasek HP, siła, licznik wrogów, modyfikator runu, loading overlay,
    \item \textbf{Menu} (\texttt{I}): Questy, Ekwipunek, Księga, Statystyki, Ustawienia,
    \item \textbf{Drag-and-drop} ekwipunku, sloty scalania i ulepszania,
    \item \textbf{Toasty zdarzeń}: ulepszenia, zapisy, zdobycia rzadkich przedmiotów, narracja, podsumowanie wyprawy,
    \item \textbf{Debug HUD AI} (\texttt{F3}),
    \item \textbf{Ekrany}: Start, Pauza, Wygrana, Przegrana.
\end{itemize}

\subsection{Feedback Walki}
\begin{itemize}
    \item Hit-stop (0{,}04$\times$ czas na 4 klatki przy trafieniu),
    \item Flash ekranu (czerwony - obrażenia, zielony - rzadki przedmiot),
    \item Drżenie kamery (\texttt{OnCameraShake}),
    \item Shader \texttt{character\_hurt} - flash trafienia oraz tint elit.
\end{itemize}

\section{Znane Ograniczenia}

Lista poniżej to nie spis wymówek, tylko mapa tego, co mam zamiar dorobić w kolejnej iteracji. Każdy punkt zostawiłem świadomie, kierując się zasadą ,,jedna rzecz mniej, ale ta jedna w pełni'':

\begin{itemize}
    \item Tylko jedna lokacja-hub (katakumby) plus proceduralny loch. Las, miasto zwierząt, ruiny i pałace zaplanowane w narracji nie weszły do tej wersji.
    \item Jeden NPC-dawca questów (Labrador Rycerz). Pełnoprawne drzewo dialogowe oraz typ celu \textit{Talk} są przygotowane w architekturze, ale czekają na uzupełnienie treścią.
    \item System rzutów kością wpływający na rozmowy nie został zaimplementowany - pojawi się razem z drzewem dialogowym.
    \item Brak warstwy audio (muzyki i efektów). FL Studio jest w planach na kolejny etap.
    \item AI elit korzysta z dyskretnej tablicy $Q$. Migracja do Deep RL (sieć neuronowa zamiast tablicy, zmienne ciągłe jako kontekst) jest naturalnym kolejnym krokiem.
    \item Kontekst bandyty ogranicza się do dwóch zmiennych (HP, liczba elit). Świadomie zostawiłem przestrzeń na rozszerzenie o ekwipunek, aktywny modyfikator runu i fazę starcia, gdy zwiększę liczbę przypadków testowych.
    \item Kolizje gracza z geometrią 3D - sporadyczne utknięcia przy slide i chód ,,w powietrzu'' na nierównościach tła (patrz rozdz.~implementacji, sekcja problemów).
    \item Elity w taktyce \textit{Kite} przy maksymalnym \texttt{PreferredDistance} mogą kręcić się w miejscu - wymaga martwej strefy na dystansie w \texttt{NavigationAgent3D}.
\end{itemize}
